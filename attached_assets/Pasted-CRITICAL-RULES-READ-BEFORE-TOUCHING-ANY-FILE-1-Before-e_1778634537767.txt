## Task 1: Remove Black Background from the Logo Image

The file `XMUM__6_.png` contains the XMUM Market logo on a **solid black background**. You need to strip away the black background so that only the logo artwork remains on a **transparent background**.

### Steps:
1. Use the `sharp` npm package (or Python's `Pillow` if the project is set up for it) to process the image.
2. Replace all pixels that are black or near-black (RGB values where R < 30, G < 30, B < 30) with full transparency (alpha = 0).
3. Save the result as `xmum-market-logo.png` inside `assets/images/` (or the equivalent assets folder in your project).
4. Verify the output visually — the logo text ("XMUM MARKET"), the orange shop icon, and the blue WiFi signal arcs should all be fully intact with no black fringing.

> **Note:** If `sharp` is unavailable, use a canvas-based approach in Node.js with the `canvas` package, or use Python with `Pillow`:
> ```python
> from PIL import Image
> import numpy as np
> img = Image.open("XMUM__6_.png").convert("RGBA")
> data = np.array(img)
> # Make near-black pixels transparent
> mask = (data[:,:,0] < 30) & (data[:,:,1] < 30) & (data[:,:,2] < 30)
> data[mask] = [0, 0, 0, 0]
> Image.fromarray(data).save("assets/images/xmum-market-logo.png")
> ```

## Task 3: Creative Loading Animation Using the Logo

Create a full-screen **splash/loading screen** that displays whenever:
1. The app first launches or the page is refreshed
2. There is a route transition that takes more than 300ms (lazy-loaded screens, data fetching, etc.)

### Animation Concept: "Market Opening"

The animation should feel like a market stall opening for the day — energetic, friendly, and on-brand.

#### Sequence (total duration ≈ 2.2 seconds):

| Step | Element | Animation | Duration |
|------|---------|-----------|----------|
| 1 | Background | Fade in from white → deep navy (`#0D2B6B`) | 0–300ms |
| 2 | Shop icon (the orange stall in the logo) | Bounce in from below, slight overshoot (spring easing) | 300–600ms |
| 3 | "XMUM" text | Slide in from the left, fade in | 500–750ms |
| 4 | "MARKET" text | Slide in from the right, fade in | 600–850ms |
| 5 | WiFi/signal arcs | Draw on one by one (staggered), each arc appears with a pulse | 750–1100ms |
| 6 | Entire logo | Subtle pulse/glow effect (scale 1.0 → 1.04 → 1.0) | 1100–1400ms |
| 7 | Loading dots | Three dots beneath the logo bounce in sequence (···) | 1100–2000ms |
| 8 | Whole screen | Fade out, reveal main app | 2000–2200ms |

#### Implementation — React Native with `Animated` API:

```tsx
// components/LoadingScreen.tsx
import React, { useEffect, useRef } from 'react';
import {
  View, Image, Animated, StyleSheet, Dimensions, Text
} from 'react-native';

const { width, height } = Dimensions.get('window');

export default function LoadingScreen({ onFinish }: { onFinish: () => void }) {
  const bgOpacity     = useRef(new Animated.Value(0)).current;
  const shopScale     = useRef(new Animated.Value(0)).current;
  const shopTranslateY= useRef(new Animated.Value(60)).current;
  const xmumOpacity   = useRef(new Animated.Value(0)).current;
  const xmumTranslateX= useRef(new Animated.Value(-40)).current;
  const marketOpacity = useRef(new Animated.Value(0)).current;
  const marketTranslateX = useRef(new Animated.Value(40)).current;
  const logoGlow      = useRef(new Animated.Value(1)).current;
  const screenOpacity = useRef(new Animated.Value(1)).current;

  // Dot bounce animations
  const dot1 = useRef(new Animated.Value(0)).current;
  const dot2 = useRef(new Animated.Value(0)).current;
  const dot3 = useRef(new Animated.Value(0)).current;

  useEffect(() => {
    Animated.sequence([
      // Step 1: background fade in
      Animated.timing(bgOpacity, { toValue: 1, duration: 300, useNativeDriver: true }),

      // Steps 2-5: logo elements appear
      Animated.parallel([
        Animated.spring(shopScale, { toValue: 1, friction: 5, tension: 80, useNativeDriver: true }),
        Animated.spring(shopTranslateY, { toValue: 0, friction: 5, tension: 80, useNativeDriver: true }),
        Animated.sequence([
          Animated.delay(200),
          Animated.parallel([
            Animated.timing(xmumOpacity,    { toValue: 1, duration: 250, useNativeDriver: true }),
            Animated.timing(xmumTranslateX, { toValue: 0, duration: 250, useNativeDriver: true }),
          ]),
        ]),
        Animated.sequence([
          Animated.delay(300),
          Animated.parallel([
            Animated.timing(marketOpacity,    { toValue: 1, duration: 250, useNativeDriver: true }),
            Animated.timing(marketTranslateX, { toValue: 0, duration: 250, useNativeDriver: true }),
          ]),
        ]),
      ]),

      // Step 6: logo pulse/glow
      Animated.sequence([
        Animated.timing(logoGlow, { toValue: 1.05, duration: 150, useNativeDriver: true }),
        Animated.timing(logoGlow, { toValue: 1,    duration: 150, useNativeDriver: true }),
      ]),
    ]).start();

    // Step 7: loading dots loop (independent)
    const bounceDot = (dot: Animated.Value, delay: number) =>
      Animated.loop(
        Animated.sequence([
          Animated.delay(delay),
          Animated.timing(dot, { toValue: -8, duration: 220, useNativeDriver: true }),
          Animated.timing(dot, { toValue:  0, duration: 220, useNativeDriver: true }),
          Animated.delay(660 - delay),
        ])
      );

    const dotAnim = Animated.parallel([
      bounceDot(dot1, 0),
      bounceDot(dot2, 220),
      bounceDot(dot3, 440),
    ]);

    setTimeout(() => {
      dotAnim.start();
      // Step 8: fade out entire screen after 2s
      setTimeout(() => {
        dotAnim.stop();
        Animated.timing(screenOpacity, {
          toValue: 0, duration: 300, useNativeDriver: true
        }).start(() => onFinish());
      }, 1000);
    }, 1100);
  }, []);

  return (
    <Animated.View style={[styles.container, { opacity: screenOpacity }]}>
      <Animated.View style={[styles.bg, { opacity: bgOpacity }]} />

      {/* Logo area */}
      <Animated.View style={{ transform: [{ scale: logoGlow }], alignItems: 'center' }}>
        {/* Shop Icon bounces in */}
        <Animated.View style={{
          transform: [{ scale: shopScale }, { translateY: shopTranslateY }],
          marginBottom: 8,
        }}>
          {/* Render just the shop emoji or an icon — replace with your actual icon asset */}
          <Text style={{ fontSize: 52 }}>🏪</Text>
        </Animated.View>

        {/* XMUM slides from left */}
        <Animated.Text style={[styles.xmumText, {
          opacity: xmumOpacity,
          transform: [{ translateX: xmumTranslateX }]
        }]}>
          XMUM
        </Animated.Text>

        {/* MARKET slides from right */}
        <Animated.Text style={[styles.marketText, {
          opacity: marketOpacity,
          transform: [{ translateX: marketTranslateX }]
        }]}>
          MARKET
        </Animated.Text>
      </Animated.View>

      {/* Loading dots */}
      <View style={styles.dotsRow}>
        {[dot1, dot2, dot3].map((dot, i) => (
          <Animated.View
            key={i}
            style={[styles.dot, { transform: [{ translateY: dot }] }]}
          />
        ))}
      </View>
    </Animated.View>
  );
}

const styles = StyleSheet.create({
  container: {
    ...StyleSheet.absoluteFillObject,
    justifyContent: 'center',
    alignItems: 'center',
    zIndex: 999,
  },
  bg: {
    ...StyleSheet.absoluteFillObject,
    backgroundColor: '#0D2B6B',
  },
  xmumText: {
    color: '#FFFFFF',
    fontSize: 28,
    fontWeight: '700',
    letterSpacing: 4,
  },
  marketText: {
    color: '#FFFFFF',
    fontSize: 52,
    fontWeight: '900',
    letterSpacing: 2,
    marginTop: -8,
  },
  dotsRow: {
    flexDirection: 'row',
    marginTop: 40,
    gap: 10,
  },
  dot: {
    width: 10,
    height: 10,
    borderRadius: 5,
    backgroundColor: '#F5A623', // orange to match the shop icon
  },
});
```
