## FEATURE 4 — Stronger password requirements on sign-up

### Current state
The password is only validated with `if (password.length < 6)` which uses `t.weakPassword`. The sign-up form has no maximum length check and no complexity requirements.

### What to implement
Industry-standard password rules for a student marketplace:
- Minimum 8 characters (increased from 6)
- Maximum 32 characters
- Must contain at least one letter AND one number

These rules apply ONLY to sign-up (new account creation). The sign-in and password-reset flows must NOT be changed — existing users may have passwords that don't meet the new rules.

### In AuthModal.tsx — sign-up validation
Find the `handleSubmit` function. In the `mode === "sign-up"` branch, replace the existing weak-password check with:

```ts

  setError(t.passwordTooShort);
  return;
}

  setError(t.passwordTooLong);
  return;

  return;
```

### In AuthModal.tsx — live password strength indicator
Below the password input in the SIGN-UP form only (not sign-in), add a visual strength bar. Add a helper function at the top of the component:

```ts
function getPasswordStrength(pw: string): { level: 0 | 1 | 2 | 3; label: string } {
  if (pw.length === 0) return { level: 0, label: "" };
  const hasLetter = /[a-zA-Z]/.test(pw);
  const hasNumber = /[0-9]/.test(pw);
  const hasSpecial = /[^a-zA-Z0-9]/.test(pw);
  const longEnough = pw.length >= 8;
  if (!longEnough || !hasLetter) return { level: 1, label: "Weak" };
  if (longEnough && hasLetter && hasNumber && !hasSpecial) return { level: 2, label: "Fair" };
  if (longEnough && hasLetter && hasNumber && hasSpecial) return { level: 3, label: "Strong" };
  return { level: 1, label: "Weak" };
}
```

Below the password `<input>` in the sign-up section, insert:
```tsx
{mode === "sign-up" && password.length > 0 && (() => {
  const { level, label } = getPasswordStrength(password);
  const colors = ["", "bg-red-400", "bg-yellow-400", "bg-green-500"];
  const widths = ["", "w-1/3", "w-2/3", "w-full"];
  return (
    <div className="mt-1.5">
      <div className="h-1 w-full bg-gray-100 dark:bg-slate-700 rounded-full overflow-hidden">
        <div className={`h-full rounded-full transition-all duration-300 ${colors[level]} ${widths[level]}`} />
      </div>
      <p className={`text-[10px] mt-0.5 ${level === 1 ? "text-red-500" : level === 2 ? "text-yellow-500" : "text-green-600"}`}>
        {label}
      </p>
    </div>
  );
})()}
```

## FEATURE 5 — Content restrictions: inappropriate language and item filter

### What to implement
A client-side content filter that runs on `handleSubmit` in both `PostPage.tsx` and `EditListingPage.tsx` BEFORE the Firestore write. It checks the title and description for prohibited content and blocks submission with a clear error message.

### Implementation — create src/lib/contentFilter.ts

Create a new file `src/lib/contentFilter.ts`:

```ts
// Client-side content filter for XMUM Marketplace.
// This is a first line of defence — not a replacement for moderation.
// Checks title and description for prohibited content before submission.

// ── Prohibited item keywords ─────────────────────────────────────────────────
// Items that should not be listed on a student marketplace.
const PROHIBITED_ITEMS = [
  // Weapons
  "weapon", "knife", "parang", "gun", "pistol", "rifle", "explosive", "bomb",
  "grenade", "ammo", "ammunition", "firearm", "blade",
  // Drugs
  "drug", "weed", "cannabis", "marijuana", "cocaine", "heroin", "meth",
  "ecstasy", "mdma", "ketamine", "ganja", "dadah",
  // Adult
  "porn", "pornography", "sex toy", "dildo", "vibrator", "adult toy",
  // Counterfeit
  "fake id", "fake ic", "counterfeit", "pirated", "bootleg",
  // Gambling
  "lottery ticket", "sports bet", "gambling",
];

// ── Harassment / inappropriate language keywords ─────────────────────────────
const INAPPROPRIATE_LANGUAGE = [
  "fuck", "shit", "asshole", "bastard", "bitch", "cunt", "dick", "pussy",
  "nigger", "nigga", "faggot", "retard", "whore", "slut",
  // Malay equivalents
  "bodoh", "babi", "puki", "lancau", "sial", "celaka", "anjing", "bangsat",
  // Chinese equivalents (pinyin)
  "tmd", "nmsl", "sb",
];

export interface FilterResult {
  passed: boolean;
  reason?: string;
}

export function checkContent(title: string, description: string): FilterResult {
  const combined = `${title} ${description}`.toLowerCase();

  for (const word of PROHIBITED_ITEMS) {
    // Use word boundary matching where possible
    const regex = new RegExp(`\\b${word.replace(/[-]/g, "[-\\s]?")}\\b`, "i");
    if (regex.test(combined)) {
      return {
        passed: false,
        reason: `Your listing appears to contain a prohibited item or keyword: "${word}". Please review our community guidelines.`,
      };
    }
  }

  for (const word of INAPPROPRIATE_LANGUAGE) {
    const regex = new RegExp(`\\b${word}\\b`, "i");
    if (regex.test(combined)) {
      return {
        passed: false,
        reason: "Your listing contains inappropriate language. Please keep the marketplace respectful.",
      };
    }
  }

  return { passed: true };
}
```

In `handleSubmit`, immediately after `setLoading(true)` and before the token refresh, add:

```ts
if (!filterResult.passed) {

  return;
}
```
