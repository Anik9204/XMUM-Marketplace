## TASK: Add rich text formatting (bold, italic, font size) to all description editors

### ARCHITECTURE DECISION — READ CAREFULLY

We are NOT using a heavyweight rich text library (no Quill, TipTap, ProseMirror, Slate, Draft.js).
We are NOT storing HTML in Firestore (XSS risk, AI moderation reads dirty text).

We ARE using a lightweight markdown-subset approach:
- Text is stored as plain-text with simple markers: `**bold**`, `*italic*`, `++large++`
- A tiny parser renders these markers into React spans at display time
- The toolbar buttons wrap selected text in these markers using the textarea selection API
- Zero new npm dependencies required

Font size options: Normal (default) and Large (`++text++`) only.
No arbitrary sizes — keeps it simple and mobile-safe.

### STEP 1: Create the shared rich text utility

Create a NEW file: `artifacts/xmum-market/src/lib/richText.ts`

```typescript
// Lightweight rich-text renderer for XMUM Market descriptions.
// Storage format: plain text with markers **bold**, *italic*, ++large++
// These markers are intentionally human-readable for AI moderation.

import React from "react";

export interface RichSegment {
  text: string;
  bold?: boolean;
  italic?: boolean;
  large?: boolean;
}

/** Strip all rich-text markers — used for plain-text previews (ListingCard, etc.) */
export function stripRichText(text: string): string {
  return text
    .replace(/\*\*(.+?)\*\*/gs, "$1")
    .replace(/\*(.+?)\*/gs, "$1")
    .replace(/\+\+(.+?)\+\+/gs, "$1");
}

/** Parse rich-text markers into segments for rendering */
export function parseRichText(text: string): RichSegment[] {
  const segments: RichSegment[] = [];
  // Combined regex: **bold**, *italic*, ++large++
  const re = /(\*\*(.+?)\*\*|\*(.+?)\*|\+\+(.+?)\+\+)/gs;
  let lastIndex = 0;
  let match: RegExpExecArray | null;

  while ((match = re.exec(text)) !== null) {
    // Plain text before this match
    if (match.index > lastIndex) {
      segments.push({ text: text.slice(lastIndex, match.index) });
    }
    const full = match[0];
    if (full.startsWith("**")) {
      segments.push({ text: match[2], bold: true });
    } else if (full.startsWith("++")) {
      segments.push({ text: match[4], large: true });
    } else {
      segments.push({ text: match[3], italic: true });
    }
    lastIndex = re.lastIndex;
  }

  // Remaining plain text
  if (lastIndex < text.length) {
    segments.push({ text: text.slice(lastIndex) });
  }

  return segments;
}

/**
 * Render rich-text segments into React elements.
 * Preserves newlines using whitespace-pre-wrap on the container.
 * Usage: <RichTextDisplay text={description} className="..." />
 */
export function RichTextDisplay({
  text,
  className = "",
}: {
  text: string;
  className?: string;
}): React.ReactElement {
  const segments = parseRichText(text);

  return React.createElement(
    "p",
    { className: `whitespace-pre-wrap leading-relaxed ${className}` },
    ...segments.map((seg, i) => {
      if (seg.bold && seg.italic) {
        return React.createElement("strong", { key: i },
          React.createElement("em", null, seg.text)
        );
      }
      if (seg.bold) return React.createElement("strong", { key: i, className: "font-semibold" }, seg.text);
      if (seg.italic) return React.createElement("em", { key: i }, seg.text);
      if (seg.large) return React.createElement("span", { key: i, className: "text-base font-medium" }, seg.text);
      return React.createElement(React.Fragment, { key: i }, seg.text);
    })
  );
}
```

### STEP 2: Create the shared RichTextEditor component

Create a NEW file: `artifacts/xmum-market/src/components/RichTextEditor.tsx`

```tsx
import React, { useRef, useCallback } from "react";
import { Bold, Italic, Type } from "lucide-react";

interface RichTextEditorProps {
  value: string;
  onChange: (val: string) => void;
  placeholder?: string;
  maxLength?: number;
  className?: string;
  autoFocus?: boolean;
}

/**
 * A plain textarea enhanced with a formatting toolbar.
 * Wraps selected text with markdown-style markers:
 *   **bold**   *italic*   ++large++
 * No external dependencies. Works on mobile and desktop.
 */
export default function RichTextEditor({
  value,
  onChange,
  placeholder = "Write something...",
  maxLength = 3000,
  className = "",
  autoFocus = false,
}: RichTextEditorProps) {
  const textareaRef = useRef<HTMLTextAreaElement>(null);

  const wrapSelection = useCallback(
    (marker: string) => {
      const el = textareaRef.current;
      if (!el) return;

      const start = el.selectionStart;
      const end = el.selectionEnd;
      const selected = value.slice(start, end);

      // If nothing selected, insert placeholder markers and place cursor between them
      if (!selected) {
        const before = value.slice(0, start);
        const after = value.slice(end);
        const insert = `${marker}${marker}`;
        const newVal = before + insert + after;
        if (newVal.length > maxLength) return;
        onChange(newVal);
        // Place cursor between the markers
        requestAnimationFrame(() => {
          el.focus();
          el.setSelectionRange(start + marker.length, start + marker.length);
        });
        return;
      }

      // Toggle: if selected text is already wrapped, unwrap it
      const before = value.slice(0, start);
      const after = value.slice(end);
      const alreadyWrapped =
        before.endsWith(marker) && after.startsWith(marker);

      if (alreadyWrapped) {
        const newVal =
          before.slice(0, before.length - marker.length) +
          selected +
          after.slice(marker.length);
        onChange(newVal);
        requestAnimationFrame(() => {
          el.focus();
          el.setSelectionRange(
            start - marker.length,
            end - marker.length
          );
        });
      } else {
        const newVal = before + marker + selected + marker + after;
        if (newVal.length > maxLength) return;
        onChange(newVal);
        requestAnimationFrame(() => {
          el.focus();
          el.setSelectionRange(
            start + marker.length,
            end + marker.length
          );
        });
      }
    },
    [value, onChange, maxLength]
  );

  const handleBold = () => wrapSelection("**");
  const handleItalic = () => wrapSelection("*");
  const handleLarge = () => wrapSelection("++");

  const handleChange = (e: React.ChangeEvent<HTMLTextAreaElement>) => {
    onChange(e.target.value.slice(0, maxLength));
  };

  return (
    <div className={`flex flex-col gap-0 ${className}`}>
      {/* Toolbar */}
      <div className="flex items-center gap-0.5 px-2 py-1.5 border-b border-gray-100 dark:border-slate-700 bg-gray-50 dark:bg-slate-800/60 rounded-t-xl">
        <ToolbarButton
          onClick={handleBold}
          title="Bold (wraps selected text in **)"
          label={<Bold size={14} strokeWidth={2.5} />}
        />
        <ToolbarButton
          onClick={handleItalic}
          title="Italic (wraps selected text in *)"
          label={<Italic size={14} strokeWidth={2} />}
        />
        <div className="w-px h-4 bg-gray-200 dark:bg-slate-600 mx-1" />
        <ToolbarButton
          onClick={handleLarge}
          title="Large text (wraps selected text in ++)"
          label={
            <span className="flex items-center gap-0.5">
              <Type size={11} strokeWidth={2} />
              <span className="text-[9px] font-bold leading-none">A</span>
            </span>
          }
        />
        <div className="flex-1" />
        {/* Hint */}
        <span className="text-[10px] text-gray-400 dark:text-slate-500 hidden sm:block pr-1">
          Select text, then tap a button to format
        </span>
      </div>

      {/* Textarea */}
      <textarea
        ref={textareaRef}
        autoFocus={autoFocus}
        value={value}
        onChange={handleChange}
        placeholder={placeholder}
        className="flex-1 w-full resize-none bg-white dark:bg-slate-900 text-gray-900 dark:text-slate-100 text-sm leading-relaxed placeholder-gray-400 dark:placeholder-slate-500 focus:outline-none px-3 py-3 min-h-[180px] rounded-b-xl"
        spellCheck
      />
    </div>
  );
}

function ToolbarButton({
  onClick,
  title,
  label,
}: {
  onClick: () => void;
  title: string;
  label: React.ReactNode;
}) {
  return (
    <button
      type="button"
      onMouseDown={(e) => {
        // Prevent textarea from losing focus/selection on mouse click
        e.preventDefault();
        onClick();
      }}
      title={title}
      className="flex items-center justify-center w-7 h-7 rounded-lg text-gray-500 dark:text-slate-400 hover:text-gray-800 dark:hover:text-slate-200 hover:bg-gray-100 dark:hover:bg-slate-700 transition-colors text-sm font-bold"
    >
      {label}
    </button>
  );
}
```

**Why `onMouseDown` with `e.preventDefault()` on toolbar buttons:** This prevents the textarea from losing its text selection when the user clicks a toolbar button. Without this, the selection is lost before the wrap logic runs. This is the standard correct approach for textarea-based rich text toolbars.
