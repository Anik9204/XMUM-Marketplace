## BUG 2 — CRITICAL: Shop not found / listings not loading on `ShopPublicPage`
 
**The problem:** In `src/pages/ShopPublicPage.tsx`, the `useEffect` that loads the shop data has a race condition. When `getShopBySlug()` returns, it sets `shop` state but then immediately calls `getShopListings()` and `getShopReviews()`. However `finally(() => setLoading(false))` runs after `getShopBySlug` resolves — before the listings/reviews finish loading. This causes a premature "Shop not found" render if the shop lookup itself takes time.
 
Also, the `getShopReviews` query in `src/lib/shops.ts` uses `orderBy("createdAt", "desc")` which requires a composite index in Firestore that may not exist yet. Until the index is built, this query silently fails and throws, causing the whole shop load to error.
 
**The fix for `ShopPublicPage.tsx`:** Replace the entire `useEffect` data loading block with this robust version:
 
```tsx
useEffect(() => {
  if (!slug) return;
  let cancelled = false;
  setLoading(true);
  setShop(null);
  setListings([]);
  setReviews([]);
 
  (async () => {
    try {
      const s = await getShopBySlug(slug);
      if (cancelled) return;
      if (!s) { setLoading(false); return; }
      setShop(s);
 
      // Load listings and reviews in parallel, but don't fail if reviews error
      const [lArr, rArr] = await Promise.allSettled([
        getShopListings(s.id),
        getShopReviews(s.id),
      ]);
      if (cancelled) return;
      if (lArr.status === "fulfilled") setListings(lArr.value);
      if (rArr.status === "fulfilled") setReviews(rArr.value);
    } catch (err) {
      console.error("[ShopPublicPage] load error:", err);
    } finally {
      if (!cancelled) setLoading(false);
    }
  })();
 
  return () => { cancelled = true; };
}, [slug]);
```
 
**The fix for `src/lib/shops.ts` — `getShopReviews`:** Replace the function with a version that handles missing index gracefully:
 
```ts
export async function getShopReviews(shopId: string): Promise<ShopReview[]> {
  try {
    const q = query(
      collection(db, "shopReviews"),
      where("shopId", "==", shopId),
      orderBy("createdAt", "desc")
    );
    const snap = await getDocs(q);
    return snap.docs.map((d) => ({ id: d.id, ...d.data() } as ShopReview));
  } catch (err: any) {
    // If index not yet built, fall back to unordered query
    if (err?.code === "failed-precondition" || err?.message?.includes("index")) {
      const q2 = query(collection(db, "shopReviews"), where("shopId", "==", shopId));
      const snap2 = await getDocs(q2);
      return snap2.docs
        .map((d) => ({ id: d.id, ...d.data() } as ShopReview))
        .sort((a, b) => b.createdAt - a.createdAt);
    }
    throw err;
  }
}
```

```ts
  const snap = await getDocs(q);
  const listings = snap.docs
    .map((d) => ({ id: d.id, ...d.data() } as ShopListing))
    .filter((l) => l.isActive !== false);
 
  // Filter out listings from inactive shops (fetch shop status in batches)
  const shopIds = [...new Set(listings.map((l) => l.shopId))];
  const shopStatuses: Record<string, boolean> = {};
  await Promise.all(
    shopIds.map(async (id) => {
      try {
        const s = await getDoc(doc(db, "shops", id));
        shopStatuses[id] = s.exists() && s.data()?.isActive !== false;
      } catch {
        shopStatuses[id] = false;
      }
    })
  );
 
  return listings
    .filter((l) => shopStatuses[l.shopId] !== false)
    .slice(0, limitCount);
}
```
 
Apply the same shop-status filter to `getShopListingsByCategory`.
