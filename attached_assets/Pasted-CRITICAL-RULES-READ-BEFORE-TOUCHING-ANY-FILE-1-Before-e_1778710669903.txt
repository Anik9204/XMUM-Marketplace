# XMUM Market — Moderation-Based Delete & Edit Protection System

## Context & Purpose

You are implementing a **report-hold protection system** for XMUM Market. The goal
is to prevent users from destroying evidence by deleting or editing reported content
before an admin can review it.

The rules are:
- If a listing has **no active reports** → owner can delete and edit it normally
  (hard delete: remove Firestore doc + Storage files).
- If a listing has **one or more pending/reviewed reports** → owner **cannot**
  delete or edit it. The listing is soft-deleted (hidden from public) if the owner
  attempts to delete, but all data and files are preserved. A blocked-action modal
  is shown explaining why with admin contact options.
- When an admin **dismisses** a report → the listing returns to fully normal state
  (visible, editable, deletable again).
- When an admin **takes action** (deletes listing) → admin performs the permanent
  delete. Report is marked actioned.
- **Account deletion** is also blocked if the user has any listings with active
  (pending or reviewed) reports.

## PHASE 2 — New file: `src/lib/reportHold.ts`

Create this file from scratch:

```typescript
import {
  collection, query, where, getDocs, updateDoc, doc, limit,
} from "firebase/firestore";
import { db } from "./firebase";

/**
 * Check if a regular listing has any active (pending or reviewed) reports.
 * Returns true if the listing should be held.
 */
export async function listingHasActiveReport(listingId: string): Promise<boolean> {
  const q = query(
    collection(db, "reports"),
    where("listingId", "==", listingId),
    where("status", "in", ["pending", "reviewed"]),
    limit(1)
  );
  const snap = await getDocs(q);
  return !snap.empty;
}

/**
 * Check if a shop listing has any active (pending or reviewed) reports.
 * ReportModal stores shopListingId in the same listingId field for shop listings.
 */
export async function shopListingHasActiveReport(listingId: string): Promise<boolean> {
  const q = query(
    collection(db, "reports"),
    where("listingId", "==", listingId),
    where("status", "in", ["pending", "reviewed"]),
    limit(1)
  );
  const snap = await getDocs(q);
  return !snap.empty;
}

/**
 * Check if a user has any listings (regular) currently under a report hold.
 * Used to block account deletion.
 */
export async function userHasHeldListings(userId: string): Promise<boolean> {
  const q = query(
    collection(db, "listings"),
    where("userId", "==", userId),
    where("isReportHeld", "==", true),
    limit(1)
  );
  const snap = await getDocs(q);
  return !snap.empty;
}

/**
 * Mark a listing as report-held (soft protection — hides from public feed).
 * Called from ReportModal after a report is successfully submitted.
 */
export async function applyReportHold(listingId: string): Promise<void> {
  await updateDoc(doc(db, "listings", listingId), {
    isReportHeld: true,
    reportHeldAt: Date.now(),
  });
}

/**
 * Mark a shop listing as report-held.
 */
export async function applyShopListingReportHold(listingId: string): Promise<void> {
  await updateDoc(doc(db, "shopListings", listingId), {
    isReportHeld: true,
    reportHeldAt: Date.now(),
  });
}

/**
 * Lift the report hold on a regular listing (called by admin on dismiss).
 */
export async function liftReportHold(listingId: string): Promise<void> {
  await updateDoc(doc(db, "listings", listingId), {
    isReportHeld: false,
    reportHeldAt: null,
  });
}

/**
 * Lift the report hold on a shop listing (called by admin on dismiss).
 */
export async function liftShopListingReportHold(listingId: string): Promise<void> {
  await updateDoc(doc(db, "shopListings", listingId), {
    isReportHeld: false,
    reportHeldAt: null,
  });
}
```

```typescript
async function handleActionAndDelete(report: ListingReport) {
  if (!isAdmin) return;
  const confirmed = window.confirm(
    `This will permanently delete the listing "${report.listingTitle}" and all its files. Continue?`
  );
  if (!confirmed) return;
  try {
    // Fetch the listing to get photo URLs before deleting
    const { getDoc } = await import("firebase/firestore");
    const { ref: storageRef, deleteObject } = await import("firebase/storage");
    const { storage } = await import("../lib/firebase");

    const listingSnap = await getDoc(doc(db, "listings", report.listingId));
    if (listingSnap.exists()) {
      const photos: string[] = listingSnap.data().photos ?? [];
      await Promise.allSettled(
        photos.map((url) => {
          const match = url.match(/\/o\/(.+?)(\?|$)/);
          if (!match) return Promise.resolve();
          const path = decodeURIComponent(match[1]);
          return deleteObject(storageRef(storage, path)).catch(() => {});
        })
      );
    }

    await deleteDoc(doc(db, "listings", report.listingId));
    await deleteDoc(doc(db, "reports", report.id));
    await addDoc(collection(db, `users/${report.listingUserId}/notifications`), {
      type: "listing_removed",
      title: "Listing removed",
      body: `Your listing "${report.listingTitle}" was removed by an admin due to a policy violation.`,
      createdAt: Date.now(),
      read: false,
    });
    if (selectedReport?.id === report.id) setSelectedReport(null);
  } catch (e) {
    console.error("[ReportsPage] delete failed:", e);
    alert("Failed to delete listing/report. Check Firestore permissions.");
  }
```
