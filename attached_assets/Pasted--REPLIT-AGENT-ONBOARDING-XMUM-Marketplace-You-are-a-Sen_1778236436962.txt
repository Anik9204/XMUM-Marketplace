## FIRESTORE SECURITY RULES
(Already deployed — do not change without explicit instruction)

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    match /users/{userId} {
      allow read: if true;
      allow write: if request.auth != null && request.auth.uid == userId;
    }
    match /listings/{listingId} {
      allow read: if true;
      allow create: if request.auth != null
                    && request.auth.token.email.matches('.*@xmu\\.edu\\.my');
      allow update, delete: if request.auth != null
                             && request.auth.uid == resource.data.userId;
    }
    match /ads/{adId} {
      allow read: if true;
      allow write: if false;
    }
  }
}

  new Promise<void>((_, reject) =>
    setTimeout(() => reject(new Error("timeout:op-name")), 6_000)
  ),
]);

// 2. All Firestore QUERIES have an index fallback
try {
  const snap = await getDocs(query(...primaryConstraints));
} catch (err: any) {

    // simpler query + client-side filter fallback
  }
  throw err;
}

// 3. Timestamp normalisation
function toMillis(val: unknown): number {
  if (typeof val === "number") return val;
  if (val && typeof (val as any).toMillis === "function") return (val as any).toMillis();
  return Date.now();
}
