```ts
export async function getListingReviews(shopListingId: string): Promise<ShopReview[]> {
  try {
    const q = query(
      collection(db, "shopReviews"),
      where("shopListingId", "==", shopListingId),
      orderBy("createdAt", "desc")
    );
    const snap = await getDocs(q);
    return snap.docs.map((d) => ({ id: d.id, ...d.data() } as ShopReview));
  } catch (err: any) {
    if (err?.code === "failed-precondition" || err?.message?.includes("index")) {
      const q2 = query(collection(db, "shopReviews"), where("shopListingId", "==", shopListingId));
      const snap2 = await getDocs(q2);
      return snap2.docs
        .map((d) => ({ id: d.id, ...d.data() } as ShopReview))
        .sort((a, b) => b.createdAt - a.createdAt);
    }
    throw err;
  }
}
```

### 4b. Add analytics query functions to `shops.ts`

```ts
// Get unique visitor count in last 30 days for a shop
export async function getShopVisitorCount30Days(shopId: string): Promise<number> {
  const since = Date.now() - 30 * 24 * 60 * 60 * 1000;
  try {
    const q = query(
      collection(db, "shopVisits"),
      where("shopId", "==", shopId),
      where("visitedAt", ">=", since)
    );
    const snap = await getDocs(q);
    return snap.size; // total visit events (not deduplicated — note to user)
  } catch {
    return 0;
  }
}

// Get per-listing view counts from shopListings (already stored as viewCount)
// and also get 30-day view events per listing
export async function getListingViews30Days(shopId: string): Promise<Record<string, number>> {
  const since = Date.now() - 30 * 24 * 60 * 60 * 1000;
  try {
    const q = query(
      collection(db, "shopVisits"),
      where("shopId", "==", shopId),
      where("visitedAt", ">=", since)
    );
    const snap = await getDocs(q);
    const counts: Record<string, number> = {};
    snap.docs.forEach((d) => {
      const lid = d.data().listingId as string;
      counts[lid] = (counts[lid] ?? 0) + 1;
    });
    return counts;
  } catch {
    return {};
  }
}
```

```tsx
function AnalyticsTab({ shopId, listings }: { shopId: string; listings: ShopListing[] }) {
  const [visitorCount, setVisitorCount] = useState<number | null>(null);
  const [listingViews, setListingViews] = useState<Record<string, number>>({});
  const [loading, setLoading] = useState(true);

  useEffect(() => {
    Promise.all([
      getShopVisitorCount30Days(shopId),
      getListingViews30Days(shopId),
    ]).then(([visitors, views]) => {
      setVisitorCount(visitors);
      setListingViews(views);
    }).finally(() => setLoading(false));
  }, [shopId]);

  if (loading) return <div className="flex justify-center py-10"><Loader2 size={22} className="animate-spin text-[#003366] dark:text-blue-400" /></div>;

  // Sort listings by 30-day views descending
  const sortedListings = [...listings].sort(
    (a, b) => (listingViews[b.id] ?? 0) - (listingViews[a.id] ?? 0)
  );

  return (
    <div className="space-y-5">
      {/* Shop visitor count */}
      <div className="bg-blue-50 dark:bg-blue-900/20 border border-blue-200 dark:border-blue-700 rounded-2xl px-5 py-4 flex items-center gap-4">
        <span className="text-3xl">👥</span>
        <div>
          <p className="text-2xl font-bold text-[#003366] dark:text-blue-300">{visitorCount ?? 0}</p>
          <p className="text-xs text-blue-600 dark:text-blue-400 font-medium">Shop page visits in the last 30 days</p>
          <p className="text-[10px] text-blue-400 dark:text-blue-500 mt-0.5">Counts every page load, not unique visitors</p>
        </div>
      </div>

      {/* Per-listing 30-day views */}
      <div>
        <h3 className="text-xs font-semibold text-gray-500 dark:text-slate-400 uppercase tracking-wide mb-3">Listing Views (Last 30 Days)</h3>
        {sortedListings.length === 0 ? (
          <p className="text-sm text-gray-400 text-center py-6">No listings yet.</p>
        ) : (
          <div className="space-y-2">
            {sortedListings.map((listing) => {
              const views30 = listingViews[listing.id] ?? 0;
              const maxViews = Math.max(...sortedListings.map((l) => listingViews[l.id] ?? 0), 1);
              const barWidth = Math.round((views30 / maxViews) * 100);
              return (
                <div key={listing.id} className="bg-white dark:bg-slate-800 rounded-xl border border-gray-100 dark:border-slate-700 px-4 py-3">
                  <div className="flex items-center justify-between mb-1.5">
                    <p className="text-xs font-semibold text-gray-800 dark:text-slate-100 truncate max-w-[70%]">{listing.title}</p>
                    <div className="text-right">
                      <span className="text-sm font-bold text-[#003366] dark:text-blue-400">{views30}</span>
                      <span className="text-[10px] text-gray-400 ml-1">views</span>
                    </div>
                  </div>
                  {/* Progress bar */}
                  <div className="w-full bg-gray-100 dark:bg-slate-700 rounded-full h-1.5">
                    <div
                      className="bg-[#003366] dark:bg-blue-500 h-1.5 rounded-full transition-all duration-500"
                      style={{ width: `${barWidth}%` }}
                    />
                  </div>
                  <div className="flex items-center justify-between mt-1">
                    <span className="text-[10px] text-gray-400">{listing.viewCount} total views all-time</span>
                    {listing.reviewCount > 0 && (
                      <span className="text-[10px] text-amber-500">⭐ {listing.rating.toFixed(1)} ({listing.reviewCount})</span>
                    )}
                  </div>
                </div>
              );
            })}
          </div>
        )}
      </div>
    </div>
  );
}
```

// Get count of pending orders + pending inquiries for a shop owner
export async function getPendingActivityCount(shopId: string): Promise<number> {

      getDocs(query(collection(db, "shopOrders"), where("shopId", "==", shopId), where("status", "==", "pending"))),
