PART 8 — Fix: Messages hard-capped at 100 with no way to load older history

THE BUG:
subscribeToMessages always queries limit(100) with no pagination. In any conversation
with more than 100 messages, all older messages are permanently invisible with no UI
indication they exist.

IMPORTANT BEFORE YOU START:
Read these files completely before touching either:
- artifacts/xmum-market/src/lib/messaging.ts
- artifacts/xmum-market/src/pages/MessagesPage.tsx

The strategy is simple and surgical: increase the real-time subscription limit to 150
(covers the vast majority of conversations without pagination complexity), and add a
separate one-time fetch function for loading older messages on demand. Then add a
"Load older messages" button in the chat UI that only appears when the subscription
returns exactly 150 messages (indicating there may be more).

export async function getOlderMessages(
  convId: string,
  oldestMessageId: string,
  oldestCreatedAt: number
): Promise<Message[]> {
  const { getDocs, query, collection, orderBy, limit, where } = await import("firebase/firestore");
  const { db } = await import("./firebase");
  const q = query(
    collection(db, "conversations", convId, "messages"),
    orderBy("createdAt", "asc"),
    where("createdAt", "<", oldestCreatedAt),
    limit(50)
  );
  const snap = await getDocs(q);
  return snap.docs.map((d) => {
    const data = d.data();
    return {
      id: d.id,
      ...data,
      createdAt: typeof data.createdAt === "number"
        ? data.createdAt
        : data.createdAt?.toMillis?.() ?? Date.now(),
      seenBy: data.seenBy ?? [],
    } as Message;
  });
}

    if (!activeConv || messages.length === 0 || loadingOlder) return;
    setLoadingOlder(true);
    try {
      const oldest = messages[0];
      const older = await getOlderMessages(activeConv.id, oldest.id, oldest.createdAt);
      if (older.length < 50) setHasOlderMessages(false);
      setMessages((prev) => {
        const existingIds = new Set(prev.map((m) => m.id));
        const newOnes = older.filter((m) => !existingIds.has(m.id));
        return [...newOnes, ...prev];
      });
    } catch {
      // silent — non-critical
    } finally {
      setLoadingOlder(false);
    }
  };
